```python
def _calculate_position_size(
    self,
    balance: Decimal,
    current_price: float,
    stop_loss_percent: Optional[Decimal] = None
) -> Decimal:
    """
    计算仓位大小（基于风险控制）
    
    仓位计算逻辑：
    1. 每笔最大亏损 = 账户总资金 × 2%
    2. 开仓价值 = 最大亏损 / 止损幅度
    3. 保证金 = 开仓价值 / 杠杆倍数
    4. 数量 = 开仓价值 / 当前价格
    """
    # 1. 计算每笔最大亏损（账户总资金 × 2%）
    max_loss = balance * Decimal('0.02')
    
    # 2. 计算开仓价值（最大亏损 / 止损幅度）
    position_value = max_loss / stop_loss_percent
    
    # 3. 计算保证金（开仓价值 / 杠杆倍数）
    margin = position_value / Decimal(str(self.leverage))
    
    # 4. 计算数量（开仓价值 / 当前价格）
    quantity = position_value / Decimal(str(current_price))
    
    return quantity
```

```python
async def _check_daily_position_limit(self) -> bool:
    """
    检查单日开仓限制
    
    逻辑：
    1. 检查当前日期是否与记录日期一致
    2. 如果日期变化，重置计数器
    3. 检查当日开仓数量是否已达上限（默认2个）
    
    Returns:
        True: 可以开仓, False: 已达上限
    """
    # 获取当前日期
    current_date = datetime.now().strftime('%Y-%m-%d')
    
    # 检查日期是否变化
    if self.daily_position_date != current_date:
        # 日期变化，重置计数器
        self.daily_position_count = 0
        self.daily_position_date = current_date
        logger.info(f"新的一天开始，重置单日开仓计数器: {current_date}")
        await self._save_state()
    
    # 获取配置的最大开仓数量
    max_daily_positions = self.config.get('trading', {}).get('max_daily_positions', 2)
    
    # 检查是否已达上限
    if self.daily_position_count >= max_daily_positions:
        logger.warning(
            f"单日开仓已达上限: {self.daily_position_count}/{max_daily_positions}",
            date=current_date
        )
        return False
    
    return True
```

```python
async def _check_max_drawdown(self) -> bool:
    """
    检查最大回撤熔断
    
    逻辑：
    1. 计算当前回撤率 = abs(累计亏损) / 初始本金
    2. 如果回撤率 >= 15%，触发熔断
    3. 熔断后暂停交易一周
    
    Returns:
        True: 可以交易, False: 触发熔断
    """
    # 检查是否在回撤熔断暂停期
    if self.drawdown_pause_until:
        if datetime.now() < self.drawdown_pause_until:
            remaining_hours = (self.drawdown_pause_until - datetime.now()).total_seconds() / 3600
            logger.warning(
                f"回撤熔断暂停中，剩余时间: {remaining_hours:.1f}小时",
                total_pnl=float(self.total_pnl),
                initial_balance=float(self.initial_balance) if self.initial_balance else 0
            )
            return False
        else:
            # 暂停期结束，重置状态
            logger.info("回撤熔断暂停期结束，恢复交易")
            self.drawdown_pause_until = None
            await self._save_state()
    
    # 检查初始本金是否设置
    if not self.initial_balance or self.initial_balance <= 0:
        logger.warning("初始本金未设置，跳过回撤检查")
        return True
    
    # 计算回撤率
    if self.total_pnl >= 0:
        # 盈利状态，无回撤
        return True
    
    drawdown = abs(self.total_pnl) / self.initial_balance
    
    # 检查是否触发熔断（回撤 >= 15%）
    if drawdown >= Decimal('0.15'):
        # 触发熔断，暂停一周
        self.drawdown_pause_until = datetime.now() + timedelta(days=7)
        
        logger.error(
            f"触发最大回撤熔断！暂停交易一周",
            drawdown_percent=float(drawdown * 100),
            total_pnl=float(self.total_pnl),
            initial_balance=float(self.initial_balance),
            resume_time=self.drawdown_pause_until.isoformat()
        )
        
        # 发送通知
        await self.notification_client.send(
            message=f"【新币做空策略熔断警告】\n累计亏损: {float(self.total_pnl):.2f} USDT\n回撤率: {float(drawdown*100):.2f}%\n初始本金: {float(self.initial_balance):.2f} USDT\n暂停交易一周\n恢复时间: {self.drawdown_pause_until.strftime('%Y-%m-%d %H:%M:%S')}",
            level="error",
            project="new_coin"
        )
        
        await self._save_state()
        return False
    
    return True
```
