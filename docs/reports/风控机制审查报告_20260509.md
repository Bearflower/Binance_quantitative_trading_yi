## 一、仓位管理机制

### 需求标准
- **每笔最大亏损**: 账户总资金 × 2%
- **杠杆**: ≤ 2倍（固定）
- **仓位计算公式**:
  ```
  最大亏损 = 账户总资金 × 2%
  开仓价值 = 最大亏损 / 止损幅度
  保证金 = 开仓价值 / 杠杆倍数
  ```

### 问题分析

1. **未根据账户总资金计算**: 使用固定保证金50 USDT，而不是账户总资金的2%

```python
def _calculate_position_size(
    self,
    balance: Decimal,
    current_price: float,
    stop_loss_percent: Decimal
) -> Decimal:
    """
    根据风险控制计算仓位大小

    Args:
        balance: 账户总资金
        current_price: 当前价格
        stop_loss_percent: 止损幅度（如0.05表示5%）

    Returns:
        仓位大小（数量）
    """
    # 1. 计算最大亏损金额（账户总资金的2%）
    max_loss = balance * Decimal('0.02')

    # 2. 计算开仓价值（最大亏损 / 止损幅度）
    position_value = max_loss / stop_loss_percent

    # 3. 计算保证金（开仓价值 / 杠杆倍数）
    margin = position_value / Decimal(str(self.leverage))

    # 4. 计算数量
    if current_price > 0:
        quantity = position_value / Decimal(str(current_price))

        logger.info(
            "仓位计算",
            balance=float(balance),
            max_loss=float(max_loss),
            stop_loss_percent=float(stop_loss_percent),
            position_value=float(position_value),
            margin=float(margin),
            quantity=float(quantity)
        )

        return quantity

    return Decimal('0')
```

```python
async def _check_max_drawdown_circuit_breaker(self) -> None:
    """
    检查最大回撤熔断

    触发条件：
    - 累计亏损 >= 15%
    """
    # 获取配置
    circuit_breaker_config = self.config.get('trading', {}).get('circuit_breaker', {})

    if not circuit_breaker_config.get('enabled', True):
        return

    max_drawdown_percent = circuit_breaker_config.get('max_drawdown_percent', 0.15)

    # 获取当前余额
    balance = await self.binance_client.get_account_balance()

            f"当前回撤",
            peak_balance=float(self.peak_balance),
            current_balance=float(current_balance),
            drawdown_percent=float(self.current_drawdown * 100)
        )

        if self.current_drawdown >= max_drawdown_percent:
```

```python
async def _add_to_stop_loss_monitor(
    self,
    symbol: str,
    entry_price: float,
    entry_time: datetime

    """添加到止损监控列表"""

        return

    check_hours = blacklist_config.get('check_hours', 24)

    # 添加到监控列表
    self.stop_loss_monitor[symbol] = {
        'entry_price': entry_price,
        'entry_time': entry_time.isoformat(),
        'monitor_until': (datetime.now() + timedelta(hours=check_hours)).isoformat()
    }

    logger.info(

        entry_price=entry_price,
        monitor_hours=check_hours
    )

    await self._save_state()

async def _check_blacklist_monitor(self) -> None:
    """检查止损监控列表，检测反向暴涨"""
    if not self.stop_loss_monitor:
        return

    blacklist_config = self.config.get('trading', {}).get('blacklist', {})
    reverse_surge_percent = blacklist_config.get('reverse_surge_percent', 0.05)

    for symbol, monitor_data in list(self.stop_loss_monitor.items()):
        try:
            # 检查是否过期
            monitor_until = datetime.fromisoformat(monitor_data['monitor_until'])
            if datetime.now() > monitor_until:
                logger.info(f"止损监控过期，移除: {symbol}")
                del self.stop_loss_monitor[symbol]
                continue

            # 获取当前价格
            ticker = await self.binance_api._request(
                "GET",
                "/fapi/v1/ticker/price",
                params={'symbol': symbol},
                signed=False
            )

            current_price = float(ticker.get('price', 0))
            entry_price = monitor_data['entry_price']

            # 计算涨幅
            price_change = (current_price - entry_price) / entry_price

            # 检查是否反向暴涨
            if price_change >= reverse_surge_percent:
                # 加入永久黑名单
                if symbol not in self.blacklist:
                    self.blacklist.append(symbol)
                    logger.warning(
                        f"币种反向暴涨，加入永久黑名单: {symbol}",
                        entry_price=entry_price,
                        current_price=current_price,
                        price_change_percent=price_change * 100
                    )

                    # 发送通知
                    await self.notification_client.send(
                        message=f"【新币黑名单警告】\n交易对: {symbol}\n入场价: {entry_price}\n当前价: {current_price}\n涨幅: {price_change*100:.2f}%\n已加入永久黑名单",
                        level="warning",
                        project="new_coin"
                    )

                # 移除监控
                del self.stop_loss_monitor[symbol]

        except Exception as e:
            logger.error(f"检查黑名单监控失败: {symbol}, 错误: {e}")

    await self._save_state()
```

```python
def should_entry(
    self,
    score_result: ScoringResult,
    three_tops_score: float,
    total_technical_score: float
) -> bool:
    """
    判断是否应该入场

    入场条件（必须同时满足）：
    1. 总分 >= 6.5
    2. 技术总分 >= 6
    3. 三次冲顶分 >= 2
    4. 无一票否决项

    Args:
        score_result: 评分结果
        three_tops_score: 三次冲顶评分
        total_technical_score: 技术面总分

    Returns:
        是否应该入场
    """
    # 检查一票否决
    if score_result.veto:
        logger.info("一票否决，不入场")
        return False

    # 检查总分
    if score_result.total_score < self.entry_threshold:
        logger.info(
            f"评分未达阈值，不入场",
            total_score=score_result.total_score,
            threshold=self.entry_threshold
        )
        return False

    # 检查技术面硬性要求（必须同时满足）
    if total_technical_score < 6.0:
        logger.info(
            f"技术总分不足，不入场",
            total_technical_score=total_technical_score,
            required=6.0
        )
        return False

    if three_tops_score < 2.0:
        logger.info(
            f"三次冲顶评分不足，不入场",
            three_tops_score=three_tops_score,
            required=2.0
        )
        return False

    logger.info(
        "满足入场条件",
        total_score=score_result.total_score,
        three_tops_score=three_tops_score,
        total_technical_score=total_technical_score
    )

    return True
```
