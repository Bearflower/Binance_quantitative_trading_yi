```bash
python docs/reviews/2026-11-26-production-regression/run_matrix.py \
  --root "$PWD" \
  --manifest docs/reviews/2026-11-26-production-regression/all-tests.json \
  --out /tmp/full-regression --jobs 4 --timeout 900
```
