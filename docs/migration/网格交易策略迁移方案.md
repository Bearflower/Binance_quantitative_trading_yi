## 1. 迁移概述

### 1.1 迁移目标

将 `Grid_Trading` 项目中的网格交易策略迁移到统一交易系统 `Binance_quantitative_trading`，实现：

- **代码复用**: 共享核心模块，减少重复代码
- **统一管理**: 统一配置、日志、监控
- **渐进迁移**: 保留原有项目作为备份，逐步迁移
- **功能对等**: 确保迁移后功能完全一致

### 1.2 迁移范围

| 模块 | 原路径 | 目标路径 | 迁移类型 |
|------|--------|----------|----------|
| 网格计算器 | `grid_trading_system/src/core/strategy/grid_calculator.py` | `strategies/grid/strategy.py` | 策略层 |
| 订单管理器 | `grid_trading_system/src/core/strategy/order_manager.py` | `strategies/grid/order_manager.py` | 策略层 |
| 持仓管理器 | `grid_trading_system/src/core/strategy/position_manager.py` | `strategies/grid/position_manager.py` | 策略层 |
| 风控管理器 | `grid_trading_system/src/core/strategy/risk_manager.py` | `strategies/grid/risk_manager.py` | 策略层 |
| K线服务 | `src/data/kline_service.py` | `shared/kline_service.py` | 共享模块 |
| 通知服务 | `src/notification/feishu_notifier.py` | `shared/notification.py` | 共享模块 |
| 交易API | `src/api/binance_api.py` | `shared/binance_api.py` | 共享模块 |
| 数据库 | `src/data/database.py` | `shared/database.py` | 共享模块 |

### 1.3 迁移策略

采用**渐进式迁移**策略，分三个阶段：

1. **第一阶段**: 共享模块迁移（K线服务、通知服务、交易API、数据库）
2. **第二阶段**: 策略核心迁移（网格计算、订单管理、持仓管理、风控管理）
3. **第三阶段**: 配置和部署迁移（配置文件、Docker、监控）

```python
# 原路径: /Users/yl/vscode/Grid_Trading/grid_trading_system/src/core/strategy/grid_calculator.py
class GridCalculator:
    """网格计算器"""
    
    def __init__(self, config: GridConfig):
        self.config = config
        
    def calculate_grid_levels(self, current_price: Decimal, 
                               volatility: Decimal = None) -> List[GridLevel]:
        """计算网格层级"""
        levels = []
        
        # 根据网格类型计算
        if self.config.grid_type == 'arithmetic':
            levels = self._calculate_arithmetic_grid(current_price)
        elif self.config.grid_type == 'geometric':
            levels = self._calculate_geometric_grid(current_price)
        elif self.config.grid_type == 'dynamic':
            levels = self._calculate_dynamic_grid(current_price, volatility)
        
        return levels
    
    def _calculate_arithmetic_grid(self, current_price: Decimal) -> List[GridLevel]:
        """计算等差网格"""
        levels = []
        grid_spacing = self.config.grid_spacing
        
        for i in range(-self.config.grid_count // 2, self.config.grid_count // 2 + 1):
            price = current_price + grid_spacing * i
            if price > 0:
                levels.append(GridLevel(
                    price=price,
                    side='BUY' if i < 0 else 'SELL',
                    quantity=self.config.base_quantity
                ))
        
        return levels
    
    def _calculate_geometric_grid(self, current_price: Decimal) -> List[GridLevel]:
        """计算等比网格"""
        levels = []
        ratio = self.config.grid_spacing_ratio
        
        for i in range(-self.config.grid_count // 2, self.config.grid_count // 2 + 1):
            price = current_price * (ratio ** i)
            levels.append(GridLevel(
                price=price,
                side='BUY' if i < 0 else 'SELL',
                quantity=self.config.base_quantity
            ))
        
        return levels
    
    def _calculate_dynamic_grid(self, current_price: Decimal, 
                                  volatility: Decimal) -> List[GridLevel]:
        """计算动态网格（基于波动率）"""
        # 根据波动率调整网格间距
        if volatility:
            adjusted_spacing = self.config.grid_spacing * (1 + volatility)
        else:
            adjusted_spacing = self.config.grid_spacing
        
        return self._calculate_arithmetic_grid(current_price)
```

**特点**:
- **多种网格类型**: 等差网格、等比网格、动态网格
- **动态调整**: 根据波动率自动调整网格间距
- **灵活配置**: 支持自定义网格数量、间距、数量

```python
class GridStrategy(BaseStrategy):

    def __init__(self, config: dict):
        super().__init__(config)
        self.grid_calculator = GridCalculator(config)
        self.order_manager = OrderManager(config)
        self.position_manager = PositionManager(config)
        self.risk_manager = RiskManager(config)
        
    def initialize_grid(self, symbol: str):
        """初始化网格"""
        # 1. 获取当前价格
        current_price = self.binance_api.get_ticker_price(symbol)
        
        # 2. 获取波动率
        volatility = self._calculate_volatility(symbol)

            current_price, volatility

        for level in grid_levels:
            self.order_manager.place_grid_order(symbol, level)
        
        # 5. 保存网格状态
        self._save_grid_state(symbol, grid_levels)
```

```python
# 原路径: /Users/yl/vscode/Grid_Trading/grid_trading_system/src/core/strategy/order_manager.py
class OrderManager:
    """订单管理器"""
    
    def __init__(self, binance_api, config):
        self.binance_api = binance_api
        self.config = config
        self.pending_orders = {}  # 待成交订单
        self.filled_orders = {}   # 已成交订单
        
    def place_grid_order(self, symbol: str, level: GridLevel) -> Order:
        """挂网格单"""
        order = self.binance_api.place_order(
            symbol=symbol,
            side=level.side,
            order_type='LIMIT',
            quantity=level.quantity,
            price=level.price
        )
        
        self.pending_orders[order['orderId']] = {
            'order': order,
            'level': level,
            'symbol': symbol
        }
        
        return order
    
    def cancel_all_orders(self, symbol: str):
        """撤销所有订单"""
        for order_id, order_info in list(self.pending_orders.items()):
            if order_info['symbol'] == symbol:
                self.binance_api.cancel_order(symbol, order_id)
                del self.pending_orders[order_id]
    
    def on_order_filled(self, order: Order):
        """订单成交回调"""
        order_id = order['orderId']
        
        if order_id in self.pending_orders:
            order_info = self.pending_orders[order_id]
            
            # 移动到已成交列表
            self.filled_orders[order_id] = order_info
            
            # 从待成交列表移除
            del self.pending_orders[order_id]
            
            # 触发反向挂单
            self._place_reverse_order(order_info)
    
    def _place_reverse_order(self, order_info: dict):
        """挂反向单"""
        level = order_info['level']
        symbol = order_info['symbol']
        
        # 计算反向价格
        reverse_price = self._calculate_reverse_price(level)
        
        # 挂反向单
        reverse_side = 'SELL' if level.side == 'BUY' else 'BUY'
        self.place_grid_order(symbol, GridLevel(
            price=reverse_price,
            side=reverse_side,
            quantity=level.quantity
        ))
```

**特点**:
- **订单跟踪**: 跟踪所有挂单状态
- **自动反向挂单**: 成交后自动挂反向单
- **批量撤销**: 支持一键撤销所有订单

```python
class OrderManager:
    """订单管理器"""
    
    def __init__(self, binance_api, db, notification, config):
        self.binance_api = binance_api
        self.db = db
        self.notification = notification
        self.config = config
        
        # 从数据库恢复订单状态
        self._restore_orders()
        
    def _restore_orders(self):
        """从数据库恢复订单状态"""
        orders = self.db.get_orders('grid', limit=100)
        for order in orders:
            if order['status'] == 'NEW':
                self.pending_orders[order['order_id']] = order
            elif order['status'] == 'FILLED':
                self.filled_orders[order['order_id']] = order
    
    def place_grid_order(self, symbol: str, level: GridLevel) -> Order:
        """挂网格单"""
        # 1. 下单
        order = self.binance_api.place_order(
            symbol=symbol,
            side=level.side,
            order_type='LIMIT',
            quantity=level.quantity,
            price=level.price
        )
        
        # 2. 保存到内存
        order_info = {
            'order': order,
            'level': level,
            'symbol': symbol
        }
        self.pending_orders[order['orderId']] = order_info
        
        # 3. 保存到数据库
        order['strategy'] = 'grid'
        self.db.save_order(order)
        
        # 4. 发送通知
        self.notification.send_message(
            f"网格单已挂出：{symbol} {level.side} {level.quantity}@{level.price}"
        )
        
        return order
```

```python
class PositionManager:
    """持仓管理器"""
    
    def __init__(self, binance_api, config):
        self.binance_api = binance_api
        self.config = config
        self.positions = {}
        
    def update_position(self, symbol: str, side: str, quantity: Decimal, price: Decimal):
        """更新持仓"""
        if symbol not in self.positions:
            self.positions[symbol] = {
                'quantity': Decimal('0'),
                'avg_price': Decimal('0'),
                'total_cost': Decimal('0')
            }
        
        pos = self.positions[symbol]
        
        if side == 'BUY':
            # 买入，增加持仓
            new_quantity = pos['quantity'] + quantity
            new_cost = pos['total_cost'] + quantity * price
            
            pos['quantity'] = new_quantity
            pos['total_cost'] = new_cost
            pos['avg_price'] = new_cost / new_quantity if new_quantity > 0 else Decimal('0')
            
        else:
            # 卖出，减少持仓
            pos['quantity'] -= quantity
            
            if pos['quantity'] <= 0:
                # 清仓
                pos['quantity'] = Decimal('0')
                pos['avg_price'] = Decimal('0')
                pos['total_cost'] = Decimal('0')
    
    def get_position_pnl(self, symbol: str, current_price: Decimal) -> Decimal:
        """计算持仓盈亏"""
        if symbol not in self.positions:
            return Decimal('0')
        
        pos = self.positions[symbol]
        
        if pos['quantity'] == 0:
            return Decimal('0')
        
        return (current_price - pos['avg_price']) * pos['quantity']
    
    def get_total_pnl(self, current_prices: Dict[str, Decimal]) -> Decimal:
        """计算总盈亏"""
        total_pnl = Decimal('0')
        
        for symbol, pos in self.positions.items():
            if symbol in current_prices:
                total_pnl += self.get_position_pnl(symbol, current_prices[symbol])
        
        return total_pnl
```

**特点**:
- **持仓跟踪**: 跟踪每个交易对的持仓
- **盈亏计算**: 实时计算持仓盈亏
- **成本计算**: 计算平均持仓成本

```python
# 原路径: /Users/yl/vscode/Grid_Trading/grid_trading_system/src/core/strategy/risk_manager.py
class RiskManager:
    """风控管理器"""
    
    def __init__(self, config):
        self.config = config
        self.max_drawdown = config.get('max_drawdown', 0.1)  # 最大回撤 10%
        self.max_position_ratio = config.get('max_position_ratio', 0.3)  # 最大仓位比例
        self.daily_loss_limit = config.get('daily_loss_limit', 0.05)  # 日亏损限制
        
        self.daily_pnl = Decimal('0')
        self.peak_value = Decimal('0')
        
    def check_risk(self, symbol: str, position: dict, 
                   current_price: Decimal, account_balance: Decimal) -> RiskResult:
        """检查风险"""
        risks = []
        
        # 1. 检查最大回撤
        current_value = account_balance + self._calculate_position_value(position, current_price)
        
        if current_value > self.peak_value:
            self.peak_value = current_value
        
        drawdown = (self.peak_value - current_value) / self.peak_value if self.peak_value > 0 else 0
        
        if drawdown > self.max_drawdown:
            risks.append(RiskAlert(
                level='HIGH',
                type='MAX_DRAWDOWN',
                message=f'回撤超过阈值：{drawdown:.2%} > {self.max_drawdown:.2%}',
                action='STOP_TRADING'
            ))
        
        # 2. 检查仓位比例
        position_value = self._calculate_position_value(position, current_price)
        position_ratio = position_value / account_balance if account_balance > 0 else 0
        
        if position_ratio > self.max_position_ratio:
            risks.append(RiskAlert(
                level='MEDIUM',
                type='POSITION_RATIO',
                message=f'仓位比例过高：{position_ratio:.2%} > {self.max_position_ratio:.2%}',
                action='REDUCE_POSITION'
            ))
        
        # 3. 检查日亏损
        if self.daily_pnl < 0 and abs(self.daily_pnl) / account_balance > self.daily_loss_limit:
            risks.append(RiskAlert(
                level='HIGH',
                type='DAILY_LOSS',
                message=f'日亏损超过限制：{abs(self.daily_pnl):.2f} USDT',
                action='STOP_TRADING'
            ))
        
        return RiskResult(
            has_risk=len(risks) > 0,
            risks=risks,
            should_stop=any(r.action == 'STOP_TRADING' for r in risks)
        )
```

**特点**:
- **多维风控**: 最大回撤、仓位比例、日亏损限制
- **风险告警**: 分级告警机制
- **自动止损**: 触发阈值自动停止交易

```python
# strategies/grid/grid_calculator.py
#!/usr/bin/env python3
"""
网格计算器
计算网格层级、价格、数量
"""

import logging
from dataclasses import dataclass
from decimal import Decimal
from typing import List, Optional

logger = logging.getLogger(__name__)


@dataclass
class GridLevel:
    """网格层级"""
    price: Decimal
    side: str  # BUY/SELL
    quantity: Decimal
    level: int = 0


@dataclass
class GridConfig:
    """网格配置"""
    grid_type: str  # arithmetic/geometric/dynamic
    grid_count: int
    grid_spacing: Decimal
    grid_spacing_ratio: Decimal
    base_quantity: Decimal
    price_range_min: Optional[Decimal] = None
    price_range_max: Optional[Decimal] = None


class GridCalculator:
    """网格计算器"""
    
    def __init__(self, config: dict):
        """
        初始化网格计算器
        
        Args:
            config: 配置字典
        """
        self.config = config
        grid_config = config.get('grid', {})
        
        self.grid_type = grid_config.get('type', 'arithmetic')
        self.grid_count = grid_config.get('count', 20)
        self.grid_spacing = Decimal(str(grid_config.get('spacing', 100)))
        self.grid_spacing_ratio = Decimal(str(grid_config.get('spacing_ratio', 1.01)))
        self.base_quantity = Decimal(str(grid_config.get('base_quantity', 0.001)))
        
        price_range = grid_config.get('price_range', {})
        self.price_range_min = Decimal(str(price_range.get('min', 0))) if price_range.get('min') else None
        self.price_range_max = Decimal(str(price_range.get('max', 0))) if price_range.get('max') else None
        
        logger.info(f"网格计算器初始化完成：类型={self.grid_type}, 数量={self.grid_count}")
    
    def calculate_grid_levels(self, current_price: Decimal, 
                               volatility: Decimal = None) -> List[GridLevel]:
        """
        计算网格层级
        
        Args:
            current_price: 当前价格
            volatility: 波动率（动态网格使用）
            
        Returns:
            网格层级列表
        """
        if self.grid_type == 'arithmetic':
            levels = self._calculate_arithmetic_grid(current_price)
        elif self.grid_type == 'geometric':
            levels = self._calculate_geometric_grid(current_price)
        elif self.grid_type == 'dynamic':
            levels = self._calculate_dynamic_grid(current_price, volatility)
        else:
            raise ValueError(f"不支持的网格类型：{self.grid_type}")
        
        # 过滤价格范围
        if self.price_range_min or self.price_range_max:
            levels = self._filter_by_price_range(levels)
        
        logger.info(f"计算网格层级完成：{len(levels)} 个层级")
        return levels
    
    def _calculate_arithmetic_grid(self, current_price: Decimal) -> List[GridLevel]:
        """计算等差网格"""
        levels = []
        half_count = self.grid_count // 2
        
        for i in range(-half_count, half_count + 1):
            price = current_price + self.grid_spacing * i
            
            if price <= 0:
                continue
            
            level = GridLevel(
                price=price,
                side='BUY' if i < 0 else ('SELL' if i > 0 else 'HOLD'),
                quantity=self.base_quantity,
                level=i + half_count
            )
            levels.append(level)
        
        return levels
    
    def _calculate_geometric_grid(self, current_price: Decimal) -> List[GridLevel]:
        """计算等比网格"""
        levels = []
        half_count = self.grid_count // 2
        
        for i in range(-half_count, half_count + 1):
            price = current_price * (self.grid_spacing_ratio ** i)
            
            level = GridLevel(
                price=price,
                side='BUY' if i < 0 else ('SELL' if i > 0 else 'HOLD'),
                quantity=self.base_quantity,
                level=i + half_count
            )
            levels.append(level)
        
        return levels
    
    def _calculate_dynamic_grid(self, current_price: Decimal, 
                                  volatility: Decimal) -> List[GridLevel]:
        """计算动态网格（基于波动率）"""
        if volatility:
            # 根据波动率调整网格间距
            adjusted_spacing = self.grid_spacing * (Decimal('1') + volatility)
        else:
            adjusted_spacing = self.grid_spacing
        
        # 临时修改间距
        original_spacing = self.grid_spacing
        self.grid_spacing = adjusted_spacing
        
        levels = self._calculate_arithmetic_grid(current_price)
        
        # 恢复原始间距
        self.grid_spacing = original_spacing
        
        return levels
    
    def _filter_by_price_range(self, levels: List[GridLevel]) -> List[GridLevel]:
        """按价格范围过滤"""
        filtered = []
        
        for level in levels:
            if self.price_range_min and level.price < self.price_range_min:
                continue
            if self.price_range_max and level.price > self.price_range_max:
                continue
            filtered.append(level)
        
        return filtered
    
    def calculate_reverse_price(self, level: GridLevel, 
                                  profit_margin: Decimal = None) -> Decimal:
        """
        计算反向挂单价格
        
        Args:
            level: 网格层级
            profit_margin: 利润边际
            
        Returns:
            反向价格
        """
        profit_margin = profit_margin or self.grid_spacing
        
        if level.side == 'BUY':
            # 买入后卖出价格
            return level.price + profit_margin
        else:
            # 卖出后买入价格
            return level.price - profit_margin
```

```python
class OrderManager:
    """订单管理器"""
    
    def __init__(self, binance_api, db, notification, config):
        """
        初始化订单管理器
        
        Args:
            binance_api: Binance API 实例
            db: 数据库服务实例
            notification: 通知服务实例
            config: 配置字典
        """

        self.db = db
        self.notification = notification
        self.config = config
        
        # 订单状态
        self.pending_orders: Dict[int, dict] = {}  # 待成交订单
        self.filled_orders: Dict[int, dict] = {}   # 已成交订单
        
        # 恢复订单状态
        self._restore_orders()
        
        logger.info(f"订单管理器初始化完成：待成交={len(self.pending_orders)}, 已成交={len(self.filled_orders)}")
    
    def _restore_orders(self):
        """从数据库恢复订单状态"""
        orders = self.db.get_orders('grid', limit=100)
        for order in orders:
            if order['status'] == 'NEW':
                self.pending_orders[order['order_id']] = order
            elif order['status'] == 'FILLED':
                self.filled_orders[order['order_id']] = order
    
    def place_grid_order(self, symbol: str, level: GridLevel) -> Optional[dict]:
        """
        挂网格单
        
        Args:
            symbol: 交易对
            level: 网格层级
            
        Returns:
            订单信息
        """
        if level.side == 'HOLD':
            return None
        
        try:
            # 1. 下单
            order = self.binance_api.place_order(
                symbol=symbol,
                side=level.side,
                order_type='LIMIT',
                quantity=level.quantity,
                price=level.price,
                time_in_force='GTC'
            )
            
            # 2. 保存到内存
            order_info = {
                'order': order,
                'level': level,
                'symbol': symbol,
                'created_at': datetime.now().isoformat()
            }
            self.pending_orders[order['orderId']] = order_info
            
            # 3. 保存到数据库
            order['strategy'] = 'grid'
            self.db.save_order(order)
            
            logger.info(f"网格单已挂出：{symbol} {level.side} {level.quantity}@{level.price}")
            
            return order
            
        except Exception as e:
            logger.error(f"挂网格单失败：{e}")
            return None
    
    def cancel_all_orders(self, symbol: str = None):
        """
        撤销所有订单
        
        Args:
            symbol: 交易对（可选，不指定则撤销所有）
        """
        cancelled_count = 0
        
        for order_id, order_info in list(self.pending_orders.items()):
            if symbol and order_info['symbol'] != symbol:
                continue
            
            try:
                self.binance_api.cancel_order(order_info['symbol'], order_id)
                del self.pending_orders[order_id]
                cancelled_count += 1
            except Exception as e:
                logger.error(f"撤销订单失败 {order_id}：{e}")
        
        logger.info(f"已撤销 {cancelled_count} 个订单")
    
    def on_order_filled(self, order: dict):
        """
        订单成交回调
        
        Args:
            order: 订单信息
        """
        order_id = order['orderId']
        
        if order_id not in self.pending_orders:
            logger.warning(f"未知订单成交：{order_id}")
            return
        
        order_info = self.pending_orders[order_id]
        level = order_info['level']
        symbol = order_info['symbol']
        
        # 移动到已成交列表
        order_info['filled_at'] = datetime.now().isoformat()
        self.filled_orders[order_id] = order_info
        
        # 从待成交列表移除
        del self.pending_orders[order_id]
        
        # 更新数据库
        order['strategy'] = 'grid'
        self.db.save_order(order)
        
        # 发送通知
        self.notification.send_message(
            f"网格单成交：{symbol} {level.side} {level.quantity}@{level.price}"
        )
        
        logger.info(f"网格单成交：{order_id}")
        
        # 返回订单信息（用于触发反向挂单）
        return order_info
    
    def get_pending_orders(self, symbol: str = None) -> List[dict]:
        """获取待成交订单"""
        if symbol:
            return [o for o in self.pending_orders.values() if o['symbol'] == symbol]
        return list(self.pending_orders.values())
    
    def get_filled_orders(self, symbol: str = None, limit: int = 100) -> List[dict]:
        """获取已成交订单"""
        orders = list(self.filled_orders.values())
        if symbol:
            orders = [o for o in orders if o['symbol'] == symbol]
        return orders[-limit:]
```

```python
"""
持仓管理器
管理网格策略的持仓和盈亏
"""

import logging
from datetime import datetime
from decimal import Decimal
from typing import Dict, Optional

logger = logging.getLogger(__name__)


class PositionManager:
    """持仓管理器"""
    
    def __init__(self, binance_api, db, config):
        """
        初始化持仓管理器
        
        Args:
            binance_api: Binance API 实例
            db: 数据库服务实例
            config: 配置字典
        """
        self.binance_api = binance_api
        self.db = db
        self.config = config
        
        # 持仓状态
        self.positions: Dict[str, dict] = {}
        
        # 恢复持仓状态
        self._restore_positions()
        
        logger.info(f"持仓管理器初始化完成：{len(self.positions)} 个持仓")
    
    def _restore_positions(self):
        """从数据库恢复持仓状态"""
        state = self.db.get_strategy_state('grid', 'positions')
        if state:
            self.positions = state.get('positions', {})
            logger.info(f"恢复持仓状态：{len(self.positions)} 个持仓")
    
    def _save_positions(self):
        """保存持仓到数据库"""
        self.db.save_strategy_state('grid', 'positions', {
            'positions': self.positions,
            'updated_at': datetime.now().isoformat()
        })
    
    def update_position(self, symbol: str, side: str, quantity: Decimal, price: Decimal):
        """
        更新持仓
        
        Args:
            symbol: 交易对
            side: 方向（BUY/SELL）
            quantity: 数量
            price: 价格
        """
        if symbol not in self.positions:
            self.positions[symbol] = {
                'quantity': Decimal('0'),
                'avg_price': Decimal('0'),
                'total_cost': Decimal('0'),
                'realized_pnl': Decimal('0'),
                'trades': 0
            }
        
        pos = self.positions[symbol]
        
        if side == 'BUY':
            # 买入，增加持仓
            new_quantity = pos['quantity'] + quantity
            new_cost = pos['total_cost'] + quantity * price
            
            pos['quantity'] = new_quantity
            pos['total_cost'] = new_cost
            pos['avg_price'] = new_cost / new_quantity if new_quantity > 0 else Decimal('0')
            
        else:
            # 卖出，减少持仓
            if pos['quantity'] > 0:
                # 计算已实现盈亏
                realized_pnl = (price - pos['avg_price']) * min(quantity, pos['quantity'])
                pos['realized_pnl'] += realized_pnl
            
            pos['quantity'] -= quantity
            
            if pos['quantity'] <= 0:
                # 清仓
                pos['quantity'] = Decimal('0')
                pos['avg_price'] = Decimal('0')
                pos['total_cost'] = Decimal('0')
        
        pos['trades'] += 1
        
        # 保存到数据库
        self._save_positions()
        
        logger.info(f"持仓更新：{symbol} {side} {quantity}@{price}, 当前持仓={pos['quantity']}")
    
    def get_position(self, symbol: str) -> Optional[dict]:
        """获取持仓"""
        return self.positions.get(symbol)
    
    def get_position_pnl(self, symbol: str, current_price: Decimal) -> Decimal:
        """
        计算持仓盈亏
        
        Args:
            symbol: 交易对
            current_price: 当前价格
            
        Returns:
            未实现盈亏
        """
        if symbol not in self.positions:
            return Decimal('0')
        
        pos = self.positions[symbol]
        
        if pos['quantity'] == 0:
            return Decimal('0')
        
        return (current_price - pos['avg_price']) * pos['quantity']
    
    def get_total_pnl(self, current_prices: Dict[str, Decimal]) -> Dict[str, Decimal]:
        """
        计算总盈亏
        
        Args:
            current_prices: 当前价格字典
            
        Returns:
            盈亏字典
        """
        total_unrealized = Decimal('0')
        total_realized = Decimal('0')
        
        for symbol, pos in self.positions.items():
            total_realized += pos['realized_pnl']
            
            if symbol in current_prices:
                total_unrealized += self.get_position_pnl(symbol, current_prices[symbol])
        
        return {
            'unrealized_pnl': total_unrealized,
            'realized_pnl': total_realized,
            'total_pnl': total_unrealized + total_realized
        }
    
    def reset_daily_pnl(self):
        """重置日盈亏"""
        for symbol in self.positions:
            self.positions[symbol]['realized_pnl'] = Decimal('0')
        self._save_positions()
```

```python
#!/usr/bin/env python3
"""
风控管理器
管理网格策略的风险控制
"""

import logging
from dataclasses import dataclass
from datetime import datetime, date
from decimal import Decimal
from typing import Dict, List, Optional

logger = logging.getLogger(__name__)


@dataclass
class RiskAlert:
    """风险告警"""
    level: str  # LOW/MEDIUM/HIGH
    type: str   # 告警类型
    message: str
    action: str  # 建议动作


@dataclass
class RiskResult:
    """风险检查结果"""
    has_risk: bool
    risks: List[RiskAlert]
    should_stop: bool


class RiskManager:
    """风控管理器"""
    
    def __init__(self, binance_api, db, notification, config):
        """
        初始化风控管理器
        
        Args:
            binance_api: Binance API 实例
            db: 数据库服务实例
            notification: 通知服务实例
            config: 配置字典
        """
        self.binance_api = binance_api
        self.db = db
        self.notification = notification
        self.config = config
        
        # 从配置读取风控参数
        risk_config = config.get('risk', {})
        self.max_drawdown = Decimal(str(risk_config.get('max_drawdown', 0.1)))
        self.max_position_ratio = Decimal(str(risk_config.get('max_position_ratio', 0.3)))
        self.daily_loss_limit = Decimal(str(risk_config.get('daily_loss_limit', 0.05)))
        
        # 状态变量
        self.peak_value = Decimal('0')
        self.daily_pnl = Decimal('0')
        self.last_check_date = date.today()
        
        # 恢复状态
        self._restore_state()
        
        logger.info(f"风控管理器初始化完成：最大回撤={self.max_drawdown}, 最大仓位={self.max_position_ratio}")
    
    def _restore_state(self):
        """从数据库恢复状态"""
        state = self.db.get_strategy_state('grid', 'risk')
        if state:
            self.peak_value = Decimal(str(state.get('peak_value', 0)))
            self.daily_pnl = Decimal(str(state.get('daily_pnl', 0)))
    
    def _save_state(self):
        """保存状态到数据库"""
        self.db.save_strategy_state('grid', 'risk', {
            'peak_value': str(self.peak_value),
            'daily_pnl': str(self.daily_pnl),
            'last_check_date': self.last_check_date.isoformat(),
            'updated_at': datetime.now().isoformat()
        })
    
    def check_risk(self, symbol: str, position: dict, 
                   current_price: Decimal, account_balance: Decimal) -> RiskResult:
        """
        检查风险
        
        Args:
            symbol: 交易对
            position: 持仓信息
            current_price: 当前价格
            account_balance: 账户余额
            
        Returns:
            风险检查结果
        """
        # 检查是否需要重置日盈亏
        today = date.today()
        if today != self.last_check_date:
            self.daily_pnl = Decimal('0')
            self.last_check_date = today
        
        risks = []
        
        # 1. 检查最大回撤
        position_value = self._calculate_position_value(position, current_price)
        current_value = account_balance + position_value
        
        if current_value > self.peak_value:
            self.peak_value = current_value
        
        if self.peak_value > 0:
            drawdown = (self.peak_value - current_value) / self.peak_value
            
            if drawdown > self.max_drawdown:
                risks.append(RiskAlert(
                    level='HIGH',
                    type='MAX_DRAWDOWN',
                    message=f'回撤超过阈值：{drawdown:.2%} > {self.max_drawdown:.2%}',
                    action='STOP_TRADING'
                ))
        
        # 2. 检查仓位比例
        if account_balance > 0:
            position_ratio = position_value / account_balance
            
            if position_ratio > self.max_position_ratio:
                risks.append(RiskAlert(
                    level='MEDIUM',
                    type='POSITION_RATIO',
                    message=f'仓位比例过高：{position_ratio:.2%} > {self.max_position_ratio:.2%}',
                    action='REDUCE_POSITION'
                ))
        
        # 3. 检查日亏损
        if self.daily_pnl < 0 and account_balance > 0:
            daily_loss_ratio = abs(self.daily_pnl) / account_balance
            
            if daily_loss_ratio > self.daily_loss_limit:
                risks.append(RiskAlert(
                    level='HIGH',
                    type='DAILY_LOSS',
                    message=f'日亏损超过限制：{abs(self.daily_pnl):.2f} USDT ({daily_loss_ratio:.2%})',
                    action='STOP_TRADING'
                ))
        
        # 发送风险告警
        if risks:
            for risk in risks:
                level = 'warning' if risk.level == 'MEDIUM' else 'error'
                self.notification.send_alert(risk.type, risk.message, level=level)
        
        # 保存状态
        self._save_state()
        
        return RiskResult(
            has_risk=len(risks) > 0,
            risks=risks,
            should_stop=any(r.action == 'STOP_TRADING' for r in risks)
        )
    
    def _calculate_position_value(self, position: dict, current_price: Decimal) -> Decimal:
        """计算持仓价值"""
        if not position:
            return Decimal('0')
        
        quantity = Decimal(str(position.get('quantity', 0)))
        return quantity * current_price
    
    def update_daily_pnl(self, pnl: Decimal):
        """更新日盈亏"""
        self.daily_pnl += pnl
        self._save_state()
```

```python
# strategies/grid/strategy.py
#!/usr/bin/env python3
"""
网格交易策略核心逻辑
"""

import logging
import time
from datetime import datetime
from decimal import Decimal
from typing import Dict, List, Optional

from grid_calculator import GridCalculator, GridLevel
from order_manager import OrderManager
from position_manager import PositionManager
from risk_manager import RiskManager

logger = logging.getLogger(__name__)


class GridStrategy:
    """网格交易策略"""
    
    def __init__(self, config: dict, binance_api, db, kline_service, notification):
        """
        初始化策略
        
        Args:
            config: 配置字典
            binance_api: Binance API 实例
            db: 数据库服务实例
            kline_service: K线服务实例
            notification: 通知服务实例
        """
        self.config = config
        self.binance_api = binance_api
        self.db = db
        self.kline_service = kline_service
        self.notification = notification
        
        self.strategy_name = config['strategy']['name']
        self.symbols = config.get('symbols', [])
        self.running = False
        
        # 初始化组件
        self.grid_calculator = GridCalculator(config)
        self.order_manager = OrderManager(binance_api, db, notification, config)
        self.position_manager = PositionManager(binance_api, db, config)
        self.risk_manager = RiskManager(binance_api, db, notification, config)
        
        # 网格状态
        self.grid_states: Dict[str, dict] = {}
        
        logger.info(f"策略初始化完成：{self.strategy_name}")
    
    def run(self):
        """运行策略"""
        self.running = True
        logger.info("策略开始运行")
        
        # 恢复策略状态
        self._restore_state()
        
        # 初始化网格
        for symbol in self.symbols:
            self._initialize_grid(symbol)
        
        # 主循环
        while self.running:
            try:
                self._execute_cycle()
                time.sleep(10)  # 每10秒检查一次
            except Exception as e:
                logger.error(f"执行周期异常：{e}")
                time.sleep(30)
    
    def stop(self):
        """停止策略"""
        self.running = False
        
        # 撤销所有订单
        for symbol in self.symbols:
            self.order_manager.cancel_all_orders(symbol)
        
        # 保存状态
        self._save_state()
        
        logger.info("策略已停止")
    
    def _initialize_grid(self, symbol: str):
        """初始化网格"""
        logger.info(f"初始化网格：{symbol}")
        
        # 1. 获取当前价格
        current_price = self.binance_api.get_ticker_price(symbol)
        
        # 2. 获取波动率
        volatility = self._calculate_volatility(symbol)
        
        # 3. 计算网格层级
        grid_levels = self.grid_calculator.calculate_grid_levels(
            current_price, volatility
        )
        
        # 4. 挂单
        for level in grid_levels:
            self.order_manager.place_grid_order(symbol, level)
        
        # 5. 保存网格状态
        self.grid_states[symbol] = {
            'current_price': str(current_price),
            'grid_levels': [{'price': str(l.price), 'side': l.side, 'quantity': str(l.quantity)} for l in grid_levels],
            'initialized_at': datetime.now().isoformat()
        }
        
        logger.info(f"网格初始化完成：{symbol}, {len(grid_levels)} 个层级")
    
    def _execute_cycle(self):
        """执行一个周期"""
        # 1. 检查订单状态
        self._check_orders()
        
        # 2. 检查风控
        self._check_risk()
        
        # 3. 监控网格状态
        self._monitor_grid()
    
    def _check_orders(self):
        """检查订单状态"""
        for symbol in self.symbols:
            # 获取未完成订单
            open_orders = self.binance_api.get_open_orders(symbol)
            pending_order_ids = set(o['orderId'] for o in open_orders)
            
            # 检查是否有订单成交
            for order_id, order_info in list(self.order_manager.pending_orders.items()):
                if order_id not in pending_order_ids:
                    # 订单已成交或取消
                    order = self._get_order_info(symbol, order_id)
                    if order and order['status'] == 'FILLED':
                        self._on_order_filled(order_info)
    
    def _on_order_filled(self, order_info: dict):
        """订单成交处理"""
        symbol = order_info['symbol']
        level = order_info['level']
        
        logger.info(f"订单成交：{symbol} {level.side} {level.quantity}@{level.price}")
        
        # 1. 更新持仓
        self.position_manager.update_position(
            symbol, level.side, level.quantity, level.price
        )
        
        # 2. 挂反向单
        reverse_price = self.grid_calculator.calculate_reverse_price(level)
        reverse_side = 'SELL' if level.side == 'BUY' else 'BUY'
        
        reverse_level = GridLevel(
            price=reverse_price,
            side=reverse_side,
            quantity=level.quantity
        )
        
        self.order_manager.place_grid_order(symbol, reverse_level)
        
        # 3. 更新风控
        pnl = self._calculate_trade_pnl(level, reverse_price)
        self.risk_manager.update_daily_pnl(pnl)
    
    def _check_risk(self):
        """检查风控"""
        for symbol in self.symbols:
            # 获取账户余额
            balance = self.binance_api.get_umfut_balance('USDT')
            
            # 获取当前价格
            current_price = self.binance_api.get_ticker_price(symbol)
            
            # 获取持仓
            position = self.position_manager.get_position(symbol)
            
            # 检查风险
            result = self.risk_manager.check_risk(
                symbol, position, current_price, balance
            )
            
            if result.should_stop:
                logger.warning(f"触发风控停止：{symbol}")
                self.order_manager.cancel_all_orders(symbol)
    
    def _monitor_grid(self):
        """监控网格状态"""
        for symbol in self.symbols:
            current_price = self.binance_api.get_ticker_price(symbol)
            
            # 检查价格是否超出网格范围
            grid_state = self.grid_states.get(symbol, {})
            grid_levels = grid_state.get('grid_levels', [])
            
            if not grid_levels:
                continue
            
            min_price = Decimal(grid_levels[0]['price'])
            max_price = Decimal(grid_levels[-1]['price'])
            
            if current_price < min_price * Decimal('0.9') or current_price > max_price * Decimal('1.1'):
                logger.warning(f"价格超出网格范围：{symbol}, 当前价格={current_price}, 范围=[{min_price}, {max_price}]")
                # 重新初始化网格
                self._reinitialize_grid(symbol)
    
    def _reinitialize_grid(self, symbol: str):
        """重新初始化网格"""
        logger.info(f"重新初始化网格：{symbol}")
        
        # 撤销所有订单
        self.order_manager.cancel_all_orders(symbol)
        
        # 重新初始化
        self._initialize_grid(symbol)
    
    def _calculate_volatility(self, symbol: str) -> Optional[Decimal]:
        """计算波动率"""
        try:
            klines = self.kline_service.get_klines(
                symbol, 
                self.config.get('kline', {}).get('interval', '1h'),
                limit=20
            )
            
            if not klines or len(klines) < 10:
                return None
            
            # 计算收益率标准差
            returns = []
            for i in range(1, len(klines)):
                close_prev = float(klines[i-1].get('close', 0))
                close_curr = float(klines[i].get('close', 0))
                if close_prev > 0:
                    returns.append((close_curr - close_prev) / close_prev)
            
            if not returns:
                return None
            
            # 计算标准差
            import statistics
            volatility = Decimal(str(statistics.stdev(returns)))
            
            return volatility
            
        except Exception as e:
            logger.error(f"计算波动率失败：{e}")
            return None
    
    def _get_order_info(self, symbol: str, order_id: int) -> Optional[dict]:
        """获取订单信息"""
        try:
            # 从 API 查询订单状态
            # 简化实现
            return None
        except Exception as e:
            logger.error(f"获取订单信息失败：{e}")
            return None
    
    def _calculate_trade_pnl(self, level: GridLevel, reverse_price: Decimal) -> Decimal:
        """计算交易盈亏"""
        if level.side == 'BUY':
            # 买入后卖出
            return (reverse_price - level.price) * level.quantity
        else:
            # 卖出后买入
            return (level.price - reverse_price) * level.quantity
    
    def _restore_state(self):
        """恢复策略状态"""
        state = self.db.get_strategy_state(self.strategy_name, 'main')
        if state:
            self.grid_states = state.get('grid_states', {})
            logger.info(f"恢复策略状态：{len(self.grid_states)} 个网格")
    
    def _save_state(self):
        """保存策略状态"""
        state = {
            'grid_states': self.grid_states,
            'last_update': datetime.now().isoformat()
        }
        self.db.save_strategy_state(self.strategy_name, 'main', state)
        logger.info("策略状态已保存")
```
