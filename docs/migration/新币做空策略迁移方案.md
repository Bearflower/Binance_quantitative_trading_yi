## 1. 迁移概述

### 1.1 迁移目标

将 `bianace_newtrade_trade` 项目中的新币做空策略迁移到统一交易系统 `Binance_quantitative_trading`，实现：

- **代码复用**: 共享核心模块，减少重复代码
- **统一管理**: 统一配置、日志、监控
- **渐进迁移**: 保留原有项目作为备份，逐步迁移
- **功能对等**: 确保迁移后功能完全一致

### 1.2 迁移范围

| 模块 | 原路径 | 目标路径 | 迁移类型 |
|------|--------|----------|----------|
| 评分引擎 | `short_selling_system/core/scoring_engine.py` | `strategies/new_coin/strategy.py` | 策略层 |
| 形态识别 | `short_selling_system/core/pattern_recognition.py` | `strategies/new_coin/pattern.py` | 策略层 |
| 新币检测 | `short_selling_system/core/listing_detector.py` | `strategies/new_coin/detector.py` | 策略层 |
| 交易执行 | `short_selling_system/core/trading_executor.py` | `strategies/new_coin/executor.py` | 策略层 |
| K线服务 | `utils/kline_service.py` | `shared/kline_service.py` | 共享模块 |
| 通知服务 | `core/notifier.py` | `shared/notification.py` | 共享模块 |
| 交易API | `core/binance_trading_api.py` | `shared/binance_api.py` | 共享模块 |
| 数据库 | `core/database.py` | `shared/database.py` | 共享模块 |

### 1.3 迁移策略

采用**渐进式迁移**策略，分三个阶段：

1. **第一阶段**: 共享模块迁移（K线服务、通知服务、交易API、数据库）
2. **第二阶段**: 策略核心迁移（评分引擎、形态识别、新币检测、交易执行）
3. **第三阶段**: 配置和部署迁移（配置文件、Docker、监控）

class PatternRecognition:
    """形态识别"""
    
    def detect_three_tops(self, klines: List[Dict]) -> Tuple[bool, float]:
        """检测三次冲顶形态"""
        # 寻找三个高点
        peaks = self._find_peaks(klines)
        
        if len(peaks) < 3:
            return False, 0.0
        
        # 检查三个高点是否接近
        last_three_peaks = peaks[-3:]
        avg_peak = sum(p['high'] for p in last_three_peaks) / 3
        
        max_deviation = max(
            abs(p['high'] - avg_peak) / avg_peak 
            for p in last_three_peaks
        )
        
        if max_deviation < 0.02:  # 偏差小于2%
            score = 4.0 if max_deviation < 0.01 else 3.0
            return True, score
        
        return False, 0.0
    
    def detect_long_upper_shadow(self, klines: List[Dict]) -> Tuple[bool, float]:
        """检测长上影线"""
        last_kline = klines[-1]
        
        body = abs(last_kline['close'] - last_kline['open'])
        upper_shadow = last_kline['high'] - max(last_kline['open'], last_kline['close'])
        
        if body > 0 and upper_shadow / body > 2.0:
            score = 3.0 if upper_shadow / body > 3.0 else 2.0
            return True, score
        
        return False, 0.0
    
    def detect_volume_divergence(self, klines: List[Dict]) -> Tuple[bool, float]:
        """检测放量滞涨"""
        recent_klines = klines[-5:]
        
        # 价格变化
        price_change = (recent_klines[-1]['close'] - recent_klines[0]['open']) / recent_klines[0]['open']
        
        # 成交量变化
        avg_volume = sum(k['volume'] for k in klines[-20:-5]) / 15
        recent_volume = sum(k['volume'] for k in recent_klines) / 5
        volume_ratio = recent_volume / avg_volume if avg_volume > 0 else 0
        
        # 放量滞涨：成交量放大但价格涨幅小
        if volume_ratio > 1.5 and abs(price_change) < 0.02:
            score = 3.0 if volume_ratio > 2.0 else 2.0
            return True, score
        
        return False, 0.0
```

**特点**:
- **三次冲顶**: 检测三个高点是否接近
- **长上影线**: 上影线/实体 > 2
- **放量滞涨**: 成交量放大但价格涨幅小

class ListingDetector:
    """新币上线检测器"""
    
    def __init__(self, binance_client):
        self.binance_client = binance_client
        self.known_symbols = set()
        
    def detect_new_listings(self) -> List[Dict]:
        """检测新上线的币种"""
        # 获取当前所有交易对
        exchange_info = self.binance_client.get_exchange_info()
        current_symbols = {s['symbol'] for s in exchange_info['symbols']}
        
        # 找出新币
        new_symbols = current_symbols - self.known_symbols
        
        # 更新已知币种
        self.known_symbols = current_symbols
        
        # 返回新币信息
        new_coins = []
        for symbol in new_symbols:
            listing_time = self._get_listing_time(symbol)
            new_coins.append({
                'symbol': symbol,
                'listing_time': listing_time,
                'listing_hours': (datetime.now() - listing_time).total_seconds() / 3600
            })
        
        return new_coins
```

**特点**:

class ListingDetector:
    """新币上线检测器"""
    
    def __init__(self, binance_api, db):
        self.binance_api = binance_api
        self.db = db
        self._load_known_symbols()
        
    def _load_known_symbols(self):
        """从数据库加载已知币种"""
        state = self.db.get_strategy_state('new_coin', 'known_symbols')
        self.known_symbols = set(state.get('symbols', [])) if state else set()
        
    def _save_known_symbols(self):
        """保存已知币种到数据库"""
        self.db.save_strategy_state('new_coin', 'known_symbols', {
            'symbols': list(self.known_symbols),
            'updated_at': datetime.now().isoformat()
        })
        
    def detect_new_listings(self) -> List[Dict]:
        """检测新上线的币种"""
        # 获取当前所有交易对
        exchange_info = self.binance_api.get_exchange_info()
        current_symbols = {s['symbol'] for s in exchange_info.get('symbols', [])}
        
        # 找出新币
        new_symbols = current_symbols - self.known_symbols
        
        # 更新已知币种
        self.known_symbols = current_symbols
        self._save_known_symbols()
        
        # 返回新币信息
        return [self._get_coin_info(symbol) for symbol in new_symbols]

# 原路径: /Users/yl/vscode/bianace_newtrade_trade/short_selling_system/core/trading_executor.py
class TradingExecutor:
    """交易执行器"""
    
    def __init__(self, binance_api, config):
        self.binance_api = binance_api
        self.config = config
        
    def execute_short(self, symbol: str, quantity: Decimal, 
                      stop_loss: Decimal = None, take_profit: Decimal = None):
        """执行做空"""
        # 1. 设置杠杆
        self.binance_api.set_leverage(symbol, self.config['leverage'])
        
        # 2. 开空仓
        order = self.binance_api.place_order(
            symbol=symbol,
            side='SELL',
            order_type='MARKET',
            quantity=quantity,
            position_side='SHORT'
        )
        
        # 3. 设置止损止盈
        if stop_loss:
            self.binance_api.place_stop_order(symbol, 'BUY', quantity, stop_loss)
        
        if take_profit:
            self.binance_api.place_take_profit_order(symbol, 'BUY', quantity, take_profit)
        
        return order
```

**特点**:
- 支持市价单和限价单
- 自动设置止损止盈
- PM 账户专用接口

class TradingExecutor:
    """交易执行器"""
    
    def __init__(self, binance_api, db, notification, config):
        self.binance_api = binance_api
        self.db = db
        self.notification = notification
        self.config = config
        
    def execute_short(self, symbol: str, score_result: Dict[str, Any]):
        """执行做空"""
        # 1. 计算仓位
        balance = self.binance_api.get_umfut_balance('USDT')
        position_size = self._calculate_position_size(balance, score_result)
        
        # 2. 获取精度
        tick_size, step_size = self.binance_api.get_symbol_precision(symbol)
        
        # 3. 格式化数量
        quantity = self._format_quantity(position_size, step_size)
        
        # 4. 设置杠杆
        self.binance_api.set_leverage(symbol, self.config['trading']['leverage'])
        
        # 5. 开空仓
        order = self.binance_api.place_order(
            symbol=symbol,
            side='SELL',
            order_type='MARKET',
            quantity=quantity,
            position_side='SHORT'
        )
        
        # 6. 保存订单
        order['strategy'] = 'new_coin'
        self.db.save_order(order)
        
        # 7. 设置止损止盈
        current_price = self.binance_api.get_ticker_price(symbol)
        self._set_stop_loss_take_profit(symbol, quantity, current_price, score_result)
        
        # 8. 发送通知
        self.notification.send_trade_notification({
            'symbol': symbol,
            'side': 'SELL',
            'quantity': str(quantity),
            'price': str(current_price),
            'status': order.get('status'),
            'score': score_result['total_score'],
            'time': datetime.now().isoformat()
        })
        
        return order

#!/usr/bin/env python3
"""
形态识别模块
检测三次冲顶、长上影线、放量滞涨等做空信号形态
"""

import logging
from typing import Dict, List, Tuple, Any

logger = logging.getLogger(__name__)


class PatternRecognizer:
    """形态识别器"""
    
    def __init__(self, config: dict = None):
        """
        初始化形态识别器
        
        Args:
            config: 配置字典
        """
        self.config = config or {}
        pattern_config = self.config.get('pattern', {})
        
        # 三次冲顶配置
        three_tops = pattern_config.get('three_tops', {})
        self.three_tops_max_deviation = three_tops.get('max_deviation', 0.02)
        self.three_tops_score_high = three_tops.get('score_high', 4.0)
        self.three_tops_score_medium = three_tops.get('score_medium', 3.0)
        
        # 长上影线配置
        long_shadow = pattern_config.get('long_upper_shadow', {})
        self.shadow_ratio_threshold = long_shadow.get('ratio_threshold', 2.0)
        self.shadow_score_high = long_shadow.get('score_high', 3.0)
        self.shadow_score_medium = long_shadow.get('score_medium', 2.0)
        
        # 放量滞涨配置
        vol_div = pattern_config.get('volume_divergence', {})
        self.volume_ratio_threshold = vol_div.get('volume_ratio_threshold', 1.5)
        self.price_change_threshold = vol_div.get('price_change_threshold', 0.02)
        self.vol_div_score_high = vol_div.get('score_high', 3.0)
        self.vol_div_score_medium = vol_div.get('score_medium', 2.0)
        
        logger.info("形态识别器初始化完成")
    
    def detect(self, klines: List[Dict]) -> Dict[str, Any]:
        """
        检测所有形态
        
        Args:
            klines: K线数据列表
            
        Returns:
            检测结果字典
        """
        if not klines or len(klines) < 2:
            return {
                'three_tops': (False, 0.0),
                'long_upper_shadow': (False, 0.0),
                'volume_divergence': (False, 0.0)
            }
        
        three_tops, three_tops_score = self.detect_three_tops(klines)
        long_shadow, long_shadow_score = self.detect_long_upper_shadow(klines)
        vol_div, vol_div_score = self.detect_volume_divergence(klines)
        
        return {
            'three_tops': (three_tops, three_tops_score),
            'long_upper_shadow': (long_shadow, long_shadow_score),
            'volume_divergence': (vol_div, vol_div_score)
        }
    
    def detect_three_tops(self, klines: List[Dict]) -> Tuple[bool, float]:
        """
        检测三次冲顶形态（自适应窗口）

        Args:
            klines: K线数据列表

        Returns:
            (是否检测到, 得分)
        """
        kline_count = len(klines)
        # 自适应窗口：K线越多检测越精确
        if kline_count >= 30:
            window = 5
        elif kline_count >= 20:
            window = 3
        else:
            window = 2

        if kline_count < 2 * window + 3:
            return False, 0.0

        peaks = self._find_peaks(klines, window)
        
        if len(peaks) < 3:
            return False, 0.0
        
        # 检查最近三个高点
        last_three_peaks = peaks[-3:]
        avg_peak = sum(p['high'] for p in last_three_peaks) / 3
        
        max_deviation = max(
            abs(p['high'] - avg_peak) / avg_peak 
            for p in last_three_peaks
        )
        
        if max_deviation < self.three_tops_max_deviation:
            score = self.three_tops_score_high if max_deviation < 0.01 else self.three_tops_score_medium
            logger.info(f"检测到三次冲顶形态：偏差={max_deviation:.4f}, 得分={score}")
            return True, score
        
        return False, 0.0
    
    def detect_long_upper_shadow(self, klines: List[Dict]) -> Tuple[bool, float]:
        """
        检测长上影线
        
        Args:
            klines: K线数据列表
            
        Returns:
            (是否检测到, 得分)
        """
        if not klines:
            return False, 0.0
        
        last_kline = klines[-1]
        
        open_price = float(last_kline.get('open', 0))
        close_price = float(last_kline.get('close', 0))
        high_price = float(last_kline.get('high', 0))
        
        body = abs(close_price - open_price)
        upper_shadow = high_price - max(open_price, close_price)
        
        if body > 0 and upper_shadow / body > self.shadow_ratio_threshold:
            score = self.shadow_score_high if upper_shadow / body > 3.0 else self.shadow_score_medium
            logger.info(f"检测到长上影线：比率={upper_shadow/body:.2f}, 得分={score}")
            return True, score
        
        return False, 0.0
    
    def detect_volume_divergence(self, klines: List[Dict]) -> Tuple[bool, float]:
        """
        检测放量滞涨（自适应基线）

        Args:
            klines: K线数据列表

        Returns:
            (是否检测到, 得分)
        """
        kline_count = len(klines)
        # 自适应：根据K线数量调整检测参数
        if kline_count >= 25:
            recent_n, baseline_start, baseline_end = 5, -20, -5
        elif kline_count >= 18:
            recent_n, baseline_start, baseline_end = 3, -kline_count, -3
        elif kline_count >= 10:
            recent_n, baseline_start, baseline_end = 2, -kline_count, -2
        else:
            return False, 0.0

        recent_klines = klines[-recent_n:]
        baseline_klines = klines[baseline_start:baseline_end]

        if len(baseline_klines) < 2:
            return False, 0.0

        first_open = float(recent_klines[0].get('open', 0))
        last_close = float(recent_klines[-1].get('close', 0))
        price_change = (last_close - first_open) / first_open if first_open > 0 else 0

        baseline_count = len(baseline_klines)
        avg_volume = sum(float(k.get('volume', 0)) for k in baseline_klines) / baseline_count
        recent_volume = sum(float(k.get('volume', 0)) for k in recent_klines) / recent_n
        volume_ratio = recent_volume / avg_volume if avg_volume > 0 else 0
        
        # 放量滞涨：成交量放大但价格涨幅小
        if volume_ratio > self.volume_ratio_threshold and abs(price_change) < self.price_change_threshold:
            score = self.vol_div_score_high if volume_ratio > 2.0 else self.vol_div_score_medium
            logger.info(f"检测到放量滞涨：量比={volume_ratio:.2f}, 价格变化={price_change:.2%}, 得分={score}")
            return True, score
        
        return False, 0.0
    
    def _find_peaks(self, klines: List[Dict], window: int = 5) -> List[Dict]:
        """
        寻找高点
        
        Args:
            klines: K线数据列表
            window: 窗口大小
            
        Returns:
            高点列表
        """
        peaks = []
        
        for i in range(window, len(klines) - window):
            current_high = float(klines[i].get('high', 0))
            
            # 检查是否是局部最高点
            is_peak = True
            for j in range(i - window, i + window + 1):
                if j != i and float(klines[j].get('high', 0)) > current_high:
                    is_peak = False
                    break
            
            if is_peak:
                peaks.append({
                    'index': i,
                    'high': current_high,
                    'time': klines[i].get('open_time')
                })
        
        return peaks
```

        if state:
            self.known_symbols = set(state.get('symbols', []))

        """保存已知币种到数据库"""

            'updated_at': datetime.now().isoformat()
        })

        """
        检测新上线的币种
        
        Returns:
            新币信息列表

            exchange_info = self.binance_api.get_exchange_info()
            current_symbols = {s['symbol'] for s in exchange_info.get('symbols', [])}
            
            # 找出新币
            new_symbols = current_symbols - self.known_symbols
            
            if new_symbols:
                logger.info(f"检测到 {len(new_symbols)} 个新币：{new_symbols}")

            self.known_symbols = current_symbols
            self._save_known_symbols()
            
            # 返回新币信息
            new_coins = []
            for symbol in new_symbols:
                coin_info = self._get_coin_info(symbol, exchange_info)
                if coin_info:
                    new_coins.append(coin_info)

                    }

            kline_service: K线服务实例
            notification: 通知服务实例

        self.db = db

        self.executor = TradingExecutor(binance_api, db, notification, config)
        
        # 当前持仓

        logger.info(f"策略初始化完成：{self.strategy_name}")

        """运行策略"""
        self.running = True
        logger.info("策略开始运行")
        
        # 恢复策略状态
        self._restore_state()
        
        # 主循环
        check_interval = self.config.get('detector', {}).get('check_interval', 300)
        
        while self.running:
            try:
                self._execute_cycle()
                time.sleep(check_interval)
            except Exception as e:
                logger.error(f"执行周期异常：{e}")
                time.sleep(60)
    
    def stop(self):
        """停止策略"""
        self.running = False
        self._save_state()
        logger.info("策略已停止")
    
    def _execute_cycle(self):
        """执行一个周期"""
        logger.info(f"开始执行周期：{datetime.now()}")
        
        # 1. 检测新币
        new_coins = self.detector.detect_new_listings()
        
        if not new_coins:
            logger.debug("未检测到新币")
            return
        
        # 2. 分析每个新币
        for coin in new_coins:
            try:
                self._analyze_coin(coin)
            except Exception as e:
                logger.error(f"分析 {coin['symbol']} 异常：{e}")
        
        # 3. 监控现有持仓
        self._monitor_positions()
    
    def _analyze_coin(self, coin: Dict[str, Any]):
        """分析单个新币"""
        symbol = coin['symbol']
        listing_hours = coin['listing_hours']
        
        logger.info(f"分析新币：{symbol}, 上线时间：{listing_hours:.1f}小时")
        
        # 检查上线时间
        detector_config = self.config.get('detector', {})
        min_hours = detector_config.get('min_listing_hours', 1)
        max_hours = detector_config.get('max_listing_hours', 48)
        
        if listing_hours < min_hours:
            logger.info(f"上线时间过短，跳过：{listing_hours:.1f}小时 < {min_hours}小时")
            return
        
        if listing_hours > max_hours:
            logger.info(f"上线时间过长，跳过：{listing_hours:.1f}小时 > {max_hours}小时")
            return
        
        # 获取K线数据
        kline_config = self.config.get('kline', {})
        klines = self.kline_service.get_klines(
            symbol, 
            kline_config.get('interval', '15m'),
            limit=kline_config.get('limit', 200)
        )
        
        if not klines or len(klines) < 50:
            logger.warning(f"K线数据不足：{symbol}")
            return
        
        # 形态识别
        patterns = self.pattern_recognizer.detect(klines)
        
        # 获取合约数据
        oi_usd = self._get_open_interest(symbol)
        total_volume = self._get_total_volume(symbol)
        funding_rate = self._get_funding_rate(symbol)
        current_price = float(klines[-1].get('close', 0))
        
        # 获取最近新币OI（用于排名）
        recent_coins_oi = self.detector.get_recent_coins_oi()
        
        # 执行评分
        result = self.scoring_engine.score(
            symbol=symbol,
            oi_usd=oi_usd,
            total_volume_usd=total_volume,
            funding_rate=funding_rate,
            three_tops_detected=patterns['three_tops'][0],
            three_tops_score=patterns['three_tops'][1],
            long_upper_shadow=patterns['long_upper_shadow'][0],
            long_upper_shadow_score=patterns['long_upper_shadow'][1],
            volume_divergence=patterns['volume_divergence'][0],
            volume_divergence_score=patterns['volume_divergence'][1],
            listing_hours=listing_hours,
            current_price=current_price,
            recent_coins_oi=recent_coins_oi
        )
        
        logger.info(f"评分结果：{symbol} - 总分={result.total_score:.1f}, 否决={result.veto}")
        
        # 检查是否应该入场
        if self.scoring_engine.should_entry(
            result, 
            patterns['three_tops'][1],
            patterns['three_tops'][1] + patterns['long_upper_shadow'][1] + patterns['volume_divergence'][1]
        ):
            self._execute_trade(symbol, result, current_price)

        max_positions = trading_config.get('max_positions', 3)
        
        if len(self.positions) >= max_positions:
            logger.warning(f"持仓数量已达上限：{len(self.positions)} >= {max_positions}")
            return
        
        # 执行做空
        order = self.executor.execute_short(symbol, score_result.to_dict())
        
        if order:
            # 记录持仓
            self.positions[symbol] = {
                'order_id': order['orderId'],
                'entry_price': current_price,
                'entry_time': datetime.now(),
                'score': score_result.total_score
            }
            
            # 保存状态
            self._save_state()
    
    def _monitor_positions(self):
        """监控现有持仓"""
        for symbol, position in list(self.positions.items()):
            try:
                # 获取当前持仓状态
                current_position = self.binance_api.get_position(symbol)
                
                if not current_position or float(current_position.get('positionAmt', 0)) == 0:
                    # 持仓已平仓
                    logger.info(f"持仓已平仓：{symbol}")
                    del self.positions[symbol]
                    self._save_state()
                    continue
                
                # 检查是否需要调整止损止盈
                # TODO: 实现动态止损止盈
                
            except Exception as e:
                logger.error(f"监控持仓异常 {symbol}：{e}")
    
    def _get_open_interest(self, symbol: str) -> float:
        """获取持仓量"""
        try:
            # 从 API 获取 OI
            # 简化实现
            return 1000000.0
        except Exception as e:
            logger.error(f"获取OI失败：{e}")
            return 0.0
    
    def _get_total_volume(self, symbol: str) -> float:
        """获取总交易量"""
        try:
            # 从数据库或 API 获取
            # 简化实现
            return 50000000.0
        except Exception as e:
            logger.error(f"获取总交易量失败：{e}")

        """获取资金费率"""

            # 从 API 获取资金费率
            # 简化实现
            return 0.0001
        except Exception as e:
            logger.error(f"获取资金费率失败：{e}")
            return 0.0

        """恢复策略状态"""
        state = self.db.get_strategy_state(self.strategy_name, 'main')
        if state:
            self.positions = state.get('positions', {})
            logger.info(f"恢复策略状态：{len(self.positions)} 个持仓")
