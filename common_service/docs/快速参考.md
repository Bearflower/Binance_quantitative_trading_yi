### 2. K 线数据获取模块 (kline_service.py)

```python
import requests
from typing import List, Dict

class KlineService:
    def __init__(self, base_url: str = "http://43.156.242.188:8765/api/v1"):
        self.base_url = base_url
    
    def get_klines(self, symbol: str, interval: str, limit: int = 100) -> List[Dict]:
        """获取 K 线数据"""
        response = requests.get(
            f"{self.base_url}/klines/latest",
            params={"symbol": symbol, "interval": interval, "limit": limit}
        )
        if response.status_code == 200:
            return response.json()["data"]
        return []
    
    def get_indicators(self, symbol: str, interval: str, period: int = 100) -> Dict:
        """获取技术指标"""
        response = requests.get(
            f"{self.base_url}/indicators",
            params={"symbol": symbol, "interval": interval, "period": period}
        )
        if response.status_code == 200:
            return response.json()["data"]
        return {}

# 使用示例
kline_service = KlineService()
klines = kline_service.get_klines("BTCUSDT", "1h")
indicators = kline_service.get_indicators("BTCUSDT", "1h")
```

### 3. 通知服务模块 (notification_service.py)

```python
import requests
from typing import Optional

class NotificationService:
    def __init__(
        self,
        base_url: str = "http://43.156.242.188:8766/api/v1",
        project: str = "btc_eth"
    ):
        self.base_url = base_url
        self.project = project
    
    def send(self, message: str, level: str = "info") -> bool:
        """发送通知"""
        try:
            response = requests.post(
                f"{self.base_url}/send",
                json={
                    "project": self.project,
                    "message": message,
                    "type": "text",
                    "level": level
                },
                timeout=5
            )
            return response.status_code == 200
        except Exception as e:
            print(f"发送通知失败：{e}")
            return False
    
    def send_info(self, message: str) -> bool:
        return self.send(message, "info")
    
    def send_warning(self, message: str) -> bool:
        return self.send(message, "warning")
    
    def send_error(self, message: str) -> bool:
        return self.send(message, "error")

# 使用示例
notifier = NotificationService(project="btc_eth")
notifier.send_info("BTC 价格上涨突破 $50000")
notifier.send_warning("注意：价格可能回调")
```
