## 使用示例

### 1. 启动调度器（自动交易）

```bash
# 启动每小时调度器（带自动交易）
python scheduler_new.py

# 立即执行一次分析（带自动交易）
python scheduler_new.py --auto-trade

# 立即执行一次分析（不带自动交易，仅测试）
python scheduler_new.py --dry-run
```

### 2. 使用核心模块

```python
from core import (
    get_signal_detector,
    calculate_position,
    get_order_generator,
    get_risk_manager
)

# 检测信号
detector = get_signal_detector()
signals = detector.detect_signals(['BTCUSDT', 'ETHUSDT', 'BNBUSDT'])

# 计算仓位
position = calculate_position(
    symbol='BTCUSDT',
    entry_price=Decimal('95000'),
    stop_loss_price=Decimal('93000'),
    direction=1,
    signal_grade='A'
)

# 生成订单
generator = get_order_generator()
template = generator.generate_order_template(...)
all_orders = generator.generate_all_orders(template)

# 风险管理
risk_manager = get_risk_manager()
margin_ratio, risk_level, need_intervention = risk_manager.check_margin_ratio(
    account_equity, used_margin
)
```

### 3. 生成报告

```python
from reporting import (
    generate_weekly_report,
    generate_monthly_report,
    generate_performance_report
)

# 生成周报
weekly_report = generate_weekly_report(trades)
print(weekly_report['summary'])

# 生成月报
monthly_report = generate_monthly_report(trades)
print(monthly_report['assessment'])

# 策略优化提醒
reminder_report = generate_performance_report(trades, period_days=30)
need_adjustment, suggestions = check_adjustment_need(reminder_report['analysis'])
```

### 4. 运行回测

```python
from backtesting import run_backtest

# 准备历史数据
historical_data = {
    'BTCUSDT': [...],  # K 线数据列表
    'ETHUSDT': [...],
    'BNBUSDT': [...]
}

# 运行回测
report = run_backtest(
    historical_data=historical_data,
    start_date=datetime(2025, 1, 1),
    end_date=datetime(2025, 12, 31),
    initial_capital=Decimal('500')
)

# 查看回测结果
print(f"总交易数：{report['summary']['total_trades']}")
print(f"总收益率：{report['summary']['total_return']:.1%}")
print(f"最大回撤：{report['summary']['max_drawdown']:.1%}")
print(f"综合评估：{report['performance_assessment']['overall']}")
```

### 5. 获取 AI 第二意见

```python
from ai_advisory import get_second_opinion, compare_decisions

# 获取 AI 意见
ai_opinion = get_second_opinion(
    signal=signal,
    market_data=market_data,
    account_status=account_status
)

# 对比决策
comparison = compare_decisions(
    system_signal=signal,
    ai_analysis=ai_opinion['ai_analysis']
)

print(f"一致性：{comparison['agreement']}")
print(f"建议：{comparison['action']}")
```
