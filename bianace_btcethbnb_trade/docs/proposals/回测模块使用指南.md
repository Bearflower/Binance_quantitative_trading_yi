# 回测模块使用指南

## 概述

回测模块允许你使用历史数据测试交易策略，评估策略在不同市场环境下的表现。

## 快速开始

### 基础示例

```python
from backtesting import run_backtest
from datetime import datetime
from decimal import Decimal

# 1. 准备历史数据
historical_data = {
    'BTCUSDT': [
        {
            'timestamp': '2025-01-01T00:00:00',
            'open': '94000',
            'high': '95000',
            'low': '93500',
            'close': '94800',
            'volume': '12345'
        },
        # ... 更多 K 线数据
    ],
    'ETHUSDT': [...],
    'BNBUSDT': [...]
}

# 2. 运行回测
report = run_backtest(
    historical_data=historical_data,
    start_date=datetime(2025, 1, 1),
    end_date=datetime(2025, 12, 31),
    initial_capital=Decimal('500')
)

# 3. 查看结果
print(f"总交易数：{report['summary']['total_trades']}")
print(f"总收益率：{report['summary']['total_return']:.1%}")
print(f"最大回撤：{report['summary']['max_drawdown']:.1%}")
print(f"综合评估：{report['performance_assessment']['overall']}")
```

### 从币安 API 获取历史数据

```python
import requests
from datetime import datetime

def fetch_binance_klines(symbol, interval='1h', limit=1000):
    """
    从币安获取历史 K 线数据
    
    Args:
        symbol: 交易对 (如 'BTCUSDT')
        interval: K 线周期 (1h, 4h, 1d)
        limit: 获取数量
    
    Returns:
        K 线数据列表
    """
    url = 'https://fapi.binance.com/fapi/v1/klines'
    params = {
        'symbol': symbol,
        'interval': interval,
        'limit': limit
    }
    
    response = requests.get(url, params=params)
    data = response.json()
    
    klines = []
    for k in data:
        kline = {
            'timestamp': datetime.fromtimestamp(k[0] / 1000).isoformat(),
            'open': str(k[1]),
            'high': str(k[2]),
            'low': str(k[3]),
            'close': str(k[4]),
            'volume': str(k[5])
        }
        klines.append(kline)
    
    return klines

# 获取多个币种的数据
historical_data = {
    'BTCUSDT': fetch_binance_klines('BTCUSDT', interval='1h', limit=2000),
    'ETHUSDT': fetch_binance_klines('ETHUSDT', interval='1h', limit=2000),
    'BNBUSDT': fetch_binance_klines('BNBUSDT', interval='1h', limit=2000)
}
```

```python
import matplotlib.pyplot as plt

def plot_equity_curve(report):
    """绘制权益曲线"""
    trades = report['trades']
    
    # 计算累计盈亏
    cumulative_pnl = [0]
    for trade in trades:
        cumulative_pnl.append(cumulative_pnl[-1] + float(trade['pnl']))
    
    # 绘制曲线
    plt.figure(figsize=(12, 6))
    plt.plot(range(len(cumulative_pnl)), cumulative_pnl)
    plt.title('Equity Curve')
    plt.xlabel('Trade Number')
    plt.ylabel('Cumulative PnL (USDT)')
    plt.grid(True)
    plt.savefig('equity_curve.png')
    plt.show()

def plot_drawdown_curve(report):
    """绘制回撤曲线"""
    trades = report['trades']
    
    # 计算回撤
    peak = 0
    drawdowns = []
    cumulative = 0
    
    for trade in trades:
        cumulative += float(trade['pnl'])
        peak = max(peak, cumulative)
        drawdown = (peak - cumulative) / peak if peak > 0 else 0
        drawdowns.append(drawdown)
    
    # 绘制曲线
    plt.figure(figsize=(12, 6))
    plt.plot(range(len(drawdowns)), drawdowns)
    plt.title('Drawdown Curve')
    plt.xlabel('Trade Number')
    plt.ylabel('Drawdown')
    plt.grid(True)
    plt.savefig('drawdown_curve.png')
    plt.show()

# 使用
plot_equity_curve(report)
plot_drawdown_curve(report)
```

### 参数优化

```python
from backtesting import StrategyBacktester
from itertools import product

def optimize_parameters(historical_data):
    """参数优化"""
    best_result = None
    best_params = None
    
    # 参数范围
    margin_range = [20, 30, 40, 50]
    leverage_s_range = [4, 5, 6]
    tp1_mult_range = [1.5, 2.0, 2.5]
    
    # 遍历所有参数组合
    for margin, lev_s, tp1 in product(margin_range, leverage_s_range, tp1_mult_range):
        # 设置参数
        params = StrategyParams()
        params.set('account.single_position_margin', Decimal(str(margin)))
        params.set('position_sizing.leverage_by_grade.S', lev_s)
        params.set('risk_management.take_profit_levels.tp1_multiplier', Decimal(str(tp1)))
        
        # 运行回测
        backtester = StrategyBacktester(params=params, initial_capital=Decimal('500'))
        report = backtester.run_backtest(
            historical_data=historical_data,
            start_date=datetime(2025, 1, 1),
            end_date=datetime(2025, 6, 30)
        )
        
        # 检查是否最优
        sharpe = report['summary']['sharpe_ratio']
        if best_result is None or sharpe > best_result['summary']['sharpe_ratio']:
            best_result = report
            best_params = {
                'margin': margin,
                'leverage_S': lev_s,
                'tp1_multiplier': tp1
            }
    
    print("最优参数:")
    print(best_params)
    print(f"\n夏普比率：{best_result['summary']['sharpe_ratio']:.2f}")
    print(f"总收益率：{best_result['summary']['total_return']:.1%}")

# 使用
optimize_parameters(historical_data)
```

```python
def backtest_multiple_periods(historical_data):
    """多周期回测"""
    periods = [
        ('2025 Q1', datetime(2025, 1, 1), datetime(2025, 3, 31)),
        ('2025 Q2', datetime(2025, 4, 1), datetime(2025, 6, 30)),
        ('2025 Q3', datetime(2025, 7, 1), datetime(2025, 9, 30)),
        ('2025 Q4', datetime(2025, 10, 1), datetime(2025, 12, 31))
    ]
    
    results = {}
    for name, start, end in periods:
        report = run_backtest(
            historical_data=historical_data,
            start_date=start,
            end_date=end,
            initial_capital=Decimal('500')
        )
        results[name] = report['summary']
    
    # 对比结果
    print("=== 多周期对比 ===\n")
    print(f"{'周期':<10} {'收益率':<12} {'胜率':<10} {'回撤':<10} {'夏普':<10}")
    print("-" * 60)
    
    for name, summary in results.items():
        print(f"{name:<10} {summary['total_return']:>10.1%} "
              f"{summary['win_rate']:>8.1%} "
              f"{summary['max_drawdown']:>8.1%} "
              f"{summary['sharpe_ratio']:>8.2f}")

# 使用
backtest_multiple_periods(historical_data)
```

## 注意事项

1. **数据质量**: 确保历史数据完整、准确
2. **滑点设置**: 实盘滑点可能大于回测设置
3. **过拟合风险**: 避免过度优化参数
4. **样本外测试**: 保留部分数据用于验证
5. **前瞻性偏差**: 确保不使用未来数据

## 常见问题

### Q1: 回测结果很好，实盘很差？
- 可能是过拟合
- 滑点和手续费设置过低
- 市场条件变化

### Q2: 如何提高回测准确性？
- 使用更长的历史数据
- 包含不同市场环境
- 合理设置滑点和手续费
- 进行样本外测试

### Q3: 回测需要多少数据？
- 至少 3-6 个月数据
- 理想是 1-2 年
- 包含牛熊周期
