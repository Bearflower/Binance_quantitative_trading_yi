# 交易结果跟踪系统 - 需求分析与技术实现方案

## 一、项目背景

当前交易系统已经具备：
- ✅ 开仓记录（trades 表）
- ✅ 持仓记录（positions 表）
- ✅ 监控日志（monitoring_logs 表，记录未实现盈亏和止盈止损触发状态）
- ✅ 账户余额快照（account_balance_snapshot 表）

**缺失的功能**：
- ❌ 平仓记录（何时平仓、平仓价格、平仓原因）
- ❌ 已实现盈亏（Realized PnL，最终盈利/亏损）
- ❌ 交易结果统计（胜率、盈亏比、平均收益率）
- ❌ 止盈止损实际触发记录（哪个价位触发、触发时间）

#### 3.2.1 平仓检测模块（position_close_detector.py）

```python
class PositionCloseDetector:
    """平仓检测器"""
    
    def __init__(self):
        self.api = get_trade_api()
        self.db = get_db_manager()
        
    def detect_close(self, symbol: str, position_side: str):
        """
        检测持仓是否已平仓
        
        检测方式：
        1. 查询当前持仓，如果持仓量为 0 则已平仓
        2. 查询币安成交历史，检测平仓订单
        3. 对比数据库中的持仓记录
        """
        # 获取当前持仓
        current_position = self.api.get_position(symbol, position_side)
        
        # 如果当前持仓为 0，但数据库中有持仓记录，则已平仓
        db_position = self.db.get_position(symbol, position_side)
        
        if db_position and (not current_position or 
                           Decimal(current_position['positionAmt']) == 0):
            # 检测到平仓，执行平仓记录流程
            self._record_close(db_position)
    
    def _record_close(self, position: Dict):
        """记录平仓信息"""
        # 1. 查询平仓订单（通过成交历史）
        close_orders = self._find_close_orders(position)
        
        # 2. 计算盈亏
        pnl_data = self._calculate_pnl(position, close_orders)
        
        # 3. 查询监控日志，获取最大浮盈/浮亏
        max_pnl_data = self._get_max_pnl_from_logs(position)
        
        # 4. 保存平仓记录
        self.db.save_closed_position({
            **position,
            **close_orders,
            **pnl_data,
            **max_pnl_data
        })
        
        # 5. 更新交易记录
        self.db.update_trade_with_close(close_orders)
        
        # 6. 更新统计数据
        self.db.update_trade_statistics()
```

#### 3.2.2 止盈止损触发检测模块（tp_sl_detector.py）

```python
class TpSlTriggerDetector:
    """止盈止损触发检测器"""
    
    def __init__(self):
        self.api = get_trade_api()
        self.db = get_db_manager()
        
    def check_triggers(self):
        """检查所有持仓的止盈止损触发情况"""
        positions = self.api.get_all_positions()
        
        for position in positions:
            # 获取该持仓的止盈止损设置
            trade = self.db.get_trade_by_position(position)
            
            if not trade:
                continue
            
            # 检查是否触及止盈/止损价
            current_price = self.api.get_ticker_price(position['symbol'])
            
            if self._is_take_profit_hit(position, trade, current_price):
                self._record_trigger(position, trade, 'TAKE_PROFIT_HIT', current_price)
            
            if self._is_stop_loss_hit(position, trade, current_price):
                self._record_trigger(position, trade, 'STOP_LOSS_HIT', current_price)
    
    def _record_trigger(self, position: Dict, trade: Dict, 
                       trigger_type: str, current_price: Decimal):
        """记录止盈止损触发"""
        self.db.save_tp_sl_trigger({
            'symbol': position['symbol'],
            'position_side': position['positionSide'],
            'trigger_type': trigger_type,
            'trigger_time': int(datetime.now().timestamp() * 1000),
            'trigger_price': current_price,
            'target_price': trade['tp_price'] if 'TAKE_PROFIT' in trigger_type 
                           else trade['sl_price'],
            'position_qty': position['positionAmt'],
            'entry_price': position['entryPrice'],
            'unrealized_profit': position['unRealizedProfit'],
            'pnl_rate': self.api.calculate_pnl_rate(position, current_price),
            'related_order_id': trade['order_id']
        })
```

#### 3.2.3 交易统计模块（trade_statistics.py）

```python
class TradeStatistics:
    """交易统计模块"""
    
    def __init__(self):
        self.db = get_db_manager()
    
    def calculate_statistics(self, 
                            period: str = 'ALL',
                            symbol: str = 'ALL',
                            position_side: str = 'ALL') -> Dict:
        """
        计算交易统计
        
        Args:
            period: 统计周期（DAILY/WEEKLY/MONTHLY/YEARLY/ALL）
            symbol: 交易对（ALL 表示全部）
            position_side: 方向（ALL 表示全部）
        
        Returns:
            统计数据字典
        """
        # 从 closed_positions 表查询数据
        closed_trades = self.db.get_closed_positions(
            period=period,
            symbol=symbol,
            position_side=position_side
        )
        
        # 计算各项指标
        stats = {
            'total_trades': len(closed_trades),
            'winning_trades': sum(1 for t in closed_trades if t['net_pnl'] > 0),
            'losing_trades': sum(1 for t in closed_trades if t['net_pnl'] < 0),
            'win_rate': 0,  # 胜率
            'total_net_pnl': sum(t['net_pnl'] for t in closed_trades),
            'avg_pnl_rate': 0,  # 平均收益率
            'profit_loss_ratio': 0,  # 盈亏比
            # ... 更多统计指标
        }
        
        return stats
    
    def generate_report(self, 
                       period: str = 'MONTHLY',
                       format: str = 'dict') -> Union[Dict, str]:
        """
        生成交易报告
        
        Args:
            period: 报告周期
            format: 输出格式（dict/json/html/markdown）
        
        Returns:
            报告数据
        """
        stats = self.calculate_statistics(period)
        
        if format == 'json':
            return json.dumps(stats, indent=2)
        elif format == 'html':
            return self._generate_html_report(stats)
        elif format == 'markdown':
            return self._generate_markdown_report(stats)
        else:
            return stats
```

class DataQueryAPI:
    """数据查询 API"""
    
    def __init__(self):
        self.db = get_db_manager()
    
    def get_trade_history(self, 
                         symbol: str = None,
                         start_date: datetime = None,
                         end_date: datetime = None,
                         limit: int = 100) -> List[Dict]:
        """查询交易历史"""
        return self.db.get_closed_positions(
            symbol=symbol,
            start_date=start_date,
            end_date=end_date,
            limit=limit
        )
    
    def get_single_trade_detail(self, order_id: int) -> Dict:
        """查询单笔交易详情"""
        # 获取开仓记录
        open_trade = self.db.get_trade_by_order_id(order_id)
        
        # 获取平仓记录
        close_record = self.db.get_closed_position_by_order_id(order_id)
        
        # 获取监控日志
        monitoring_logs = self.db.get_monitoring_logs_by_order_id(order_id)
        
        # 获取止盈止损触发记录
        tp_sl_triggers = self.db.get_tp_sl_triggers_by_order_id(order_id)
        
        return {
            'open_trade': open_trade,
            'close_record': close_record,
            'monitoring_logs': monitoring_logs,
            'tp_sl_triggers': tp_sl_triggers,
            'summary': self._generate_trade_summary(open_trade, close_record)
        }
    
    def get_statistics(self,
                      period: str = 'MONTHLY',
                      symbol: str = None) -> Dict:
        """查询统计数据"""
        return self.db.get_trade_statistics(period, symbol)
    
    def export_data(self,
                   data_type: str = 'trades',
                   format: str = 'csv',
                   filters: Dict = None) -> str:
        """导出数据"""
        data = self._query_data(data_type, filters)
        
        if format == 'csv':
            return self._export_to_csv(data)
        elif format == 'excel':
            return self._export_to_excel(data)
        elif format == 'json':
            return json.dumps(data, indent=2)
