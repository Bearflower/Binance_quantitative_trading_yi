```python
def _update_trade_statistics(self, symbol: str, pnl: Decimal, trade_date: datetime.date):
    """更新交易统计（盈亏）"""
    is_win = pnl > 0
    
    if is_win:
        # 更新盈利计数
        UPDATE daily_execution_stats SET win_count = win_count + 1 ...
    else:
        # 更新亏损计数
        UPDATE daily_execution_stats SET loss_count = loss_count + 1 ...
```
